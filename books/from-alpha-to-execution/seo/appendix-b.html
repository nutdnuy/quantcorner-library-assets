B.10 Safety: Pre-trade Gate, Kill Switch และ Safe State

             At (o) = 1{price valid}1{size valid}1{position valid}

         safe state ต้องระบุเป็น action ที่ตรวจได้ เช่น

def step(event, state, alpha_engine, portfolio, risk, execution):
  validate_top(event)
  state = state.with_market(event)


  alpha = alpha_engine.on_market(event, state)
  raw_target = portfolio.target(alpha, state)
  decision = risk.check(raw_target, state)

  orders = ()
  if decision.approved:
     orders = execution.orders(decision.target, state.book)

  return state.record(alpha, decision, orders)
