def hac_mean_test(x, maxlags=5):
  x = pd.Series(x).dropna().astype(float)
  X = np.ones((len(x), 1))
  model = sm.OLS(x.to_numpy(), X).fit(
      cov_type="HAC", cov_kwds={"maxlags": maxlags}
  )
  return {
      "mean": float(model.params[0]),
      "se_hac": float(model.bse[0]),
      "t_hac": float(model.tvalues[0]),
      "p_hac": float(model.pvalues[0]),
  }



def bh_fdr(pvalues, q=0.05):
  p = np.asarray(pvalues, dtype=float)
  reject, p_adj, _, _ = multipletests(p, alpha=q, method="fdr_bh")
  return reject, p_adj



def bayes_shrink(mu_hat, se, tau):
  kappa = tau**2 / (tau**2 + se**2)
  mu_post = kappa * mu_hat
  var_post = (tau**2 * se**2) / (tau**2 + se**2)
  return mu_post, np.sqrt(var_post), kappa
